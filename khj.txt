
#include <SPI.h>
#include <MFRC522.h>
#include <Keypad.h>
#include <Servo.h>

// RFID 설정
#define SS_PIN 10
#define RST_PIN 9

MFRC522 mfrc522(SS_PIN, RST_PIN);

// 서보모터
Servo lockServo;
#define SERVO_PIN 6

// LED 및 부저
#define RED_LED 8
#define GREEN_LED 7
#define BUZZER A0

// 키패드 설정
const byte ROWS = 4;
const byte COLS = 4;

char keys[ROWS][COLS] = {
  {'1','2','3','A'},
  {'4','5','6','B'},
  {'7','8','9','C'},
  {'*','0','#','D'}
};

// 기존 배선 유지
byte rowPins[ROWS] = {A1, A2, A3, 0};
byte colPins[COLS] = {2, 3, 4, 5};

Keypad keypad = Keypad(
  makeKeymap(keys),
  rowPins,
  colPins,
  ROWS,
  COLS
);

// 비밀번호
String password = "1234";
String inputPassword = "";

// 실제 RFID 카드 UID로 변경
byte validUID[4] = {0x01, 0x02, 0x03, 0x04};

void setup()
{
  Serial.begin(9600);

  SPI.begin();
  mfrc522.PCD_Init();

  pinMode(RED_LED, OUTPUT);
  pinMode(GREEN_LED, OUTPUT);
  pinMode(BUZZER, OUTPUT);

  lockServo.attach(SERVO_PIN);
  lockServo.write(0);

  digitalWrite(RED_LED, HIGH);
  digitalWrite(GREEN_LED, LOW);

  Serial.println("SMART DOOR LOCK SYSTEM");
  Serial.println("Scan Card or Enter Password");
}

void loop()
{
  checkRFID();
  checkPassword();
}

// RFID 카드 확인
void checkRFID()
{
  if (!mfrc522.PICC_IsNewCardPresent())
    return;

  if (!mfrc522.PICC_ReadCardSerial())
    return;

  bool access = true;

  if (mfrc522.uid.size != 4)
  {
    access = false;
  }
  else
  {
    for (byte i = 0; i < 4; i++)
    {
      if (mfrc522.uid.uidByte[i] != validUID[i])
      {
        access = false;
      }
    }
  }

  if (access)
  {
    Serial.println("RFID Accepted");
    unlockDoor();
  }
  else
  {
    Serial.println("Access Denied");
    denied();
  }

  mfrc522.PICC_HaltA();
  mfrc522.PCD_StopCrypto1();
}

// 키패드 비밀번호 확인
void checkPassword()
{
  char key = keypad.getKey();

  if (key)
  {
    Serial.print("Key pressed: ");
    Serial.println(key);

    if (key == '#')
    {
      if (inputPassword == password)
      {
        Serial.println("Correct Password");
        unlockDoor();
      }
      else
      {
        Serial.println("Wrong Password");
        denied();
      }

      inputPassword = "";
    }

    else if (key == '*')
    {
      inputPassword = "";
      Serial.println("Password Cleared");
    }

    else
    {
      inputPassword += key;

      Serial.print("Input: ");
      for (int i = 0; i < inputPassword.length(); i++)
      {
        Serial.print("*");
      }
      Serial.println();
    }
  }
}

// 문 열기
void unlockDoor()
{
  digitalWrite(RED_LED, LOW);
  digitalWrite(GREEN_LED, HIGH);

  tone(BUZZER, 1000, 200);

  Serial.println("Door Open");

  lockServo.write(90);

  delay(5000);

  lockServo.write(0);

  digitalWrite(GREEN_LED, LOW);
  digitalWrite(RED_LED, HIGH);

  Serial.println("Door Locked");

  delay(1000);
}

// 접근 거부
void denied()
{
  digitalWrite(GREEN_LED, LOW);
  digitalWrite(RED_LED, HIGH);

  Serial.println("Access Denied!");

  for (int i = 0; i < 3; i++)
  {
    tone(BUZZER, 500);
    delay(250);

    noTone(BUZZER);
    delay(250);
  }
}
